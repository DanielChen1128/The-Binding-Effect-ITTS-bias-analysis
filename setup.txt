# BindingBias Environment Setup Guide
# Complete setup instructions for TTS bias research environment
# The Binding Effect: Multi-Dimensional Gender Bias in Instruction TTS
# ============================================================

# ============================================================
# Step 1: Create Conda Environment
# ============================================================
conda create -n BindingBias python=3.9.18 -y
conda activate BindingBias

# ============================================================
# Step 2: Set CUDA Environment Variables
# ============================================================
# Point CUDA_HOME at your local CUDA 11.8 installation.
export CUDA_HOME=/path/to/cuda-11.8
export PATH=$CUDA_HOME/bin:$PATH
export LD_LIBRARY_PATH=$CUDA_HOME/lib64:$LD_LIBRARY_PATH

# Optionally add to ~/.bashrc for persistence:
# echo 'export CUDA_HOME=/path/to/cuda-11.8' >> ~/.bashrc
# echo 'export PATH=$CUDA_HOME/bin:$PATH' >> ~/.bashrc
# echo 'export LD_LIBRARY_PATH=$CUDA_HOME/lib64:$LD_LIBRARY_PATH' >> ~/.bashrc

# ============================================================
# Step 3: Install PyTorch with CUDA 11.8 Support
# ============================================================
pip install torch==2.3.0 torchvision==0.18.0 torchaudio==2.3.0 --index-url https://download.pytorch.org/whl/cu118

# ============================================================
# Step 4: Downgrade pip (required for fairseq compatibility)
# ============================================================
pip install --upgrade pip==24.0

# ============================================================
# Step 5: Install Standard Requirements
# ============================================================
pip install -r requirements.txt

# ============================================================
# Step 6: Install Special Git-based Packages
# ============================================================


# Install parler-tts from GitHub
pip install git+https://github.com/huggingface/parler-tts.git

# ============================================================
# Step 7: Install Additional Dependencies
# ============================================================

# Install encodec
pip install encodec

# Install transformers (must be <4.46 for torch 2.3 compatibility)
pip install 'transformers<4.46'

# Install fairseq from GitHub (avoids pip metadata issues)
pip install fairseq==0.6.2

pip install --no-cache-dir cffi soundfile

pip install tqdm

pip install "numpy<1.24"
